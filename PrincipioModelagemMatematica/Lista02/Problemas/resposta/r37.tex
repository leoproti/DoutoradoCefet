\textbf{Fórmula do valor presente:}

O valor presente $PV$ de um montante futuro $FV$ após $n$ anos com taxa anual $r$ é:
\[
PV = \frac{FV}{(1+r)^n}.
\]

Dados: $FV=\$1{.}000{.}000$, $r=5{,}5\%=0{,}055$.

\textbf{Para 2022 (depositado em 2002, $n=20$ anos):}
\[
(1{,}055)^{20}=e^{20\ln(1{,}055)}=e^{20\times0{,}05354}=e^{1{,}0707}\approx2{,}9178.
\]
\[
PV_{2022}=\frac{1{.}000{.}000}{2{,}9178}\approx\mathbf{\$342{.}729}.
\]

\textbf{Para 2042 (depositado em 2002, $n=40$ anos):}
\[
(1{,}055)^{40}=\left[(1{,}055)^{20}\right]^2\approx(2{,}9178)^2\approx8{,}5136.
\]
\[
PV_{2042}=\frac{1{.}000{.}000}{8{,}5136}\approx\mathbf{\$117{.}463}.
\]

\textbf{Resumo:}
\begin{center}
\begin{tabular}{lll}
\hline
Meta & Anos & Valor a depositar em 2002 \\
\hline
\$1.000.000 em 2022 & 20 & \$342.729 \\
\$1.000.000 em 2042 & 40 & \$117.463 \\
\hline
\end{tabular}
\end{center}

\textbf{Interpretação:} quanto maior o horizonte de investimento, menor o valor necessário agora, devido ao efeito dos juros compostos ao longo do tempo. Dobrar o prazo (de 20 para 40 anos) reduz o depósito necessário em aproximadamente $66\%$.
