\textbf{Séries de Taylor (dois termos) para $\sin x$ e $\cos x$:}
\[
\sin x = \sum_{n=0}^{\infty}\frac{(-1)^n x^{2n+1}}{(2n+1)!}
= x - \frac{x^3}{6} + \frac{x^5}{120} - \cdots \approx x - \frac{x^3}{6},
\]
\[
\cos x = \sum_{n=0}^{\infty}\frac{(-1)^n x^{2n}}{(2n)!}
= 1 - \frac{x^2}{2} + \frac{x^4}{24} - \cdots \approx 1 - \frac{x^2}{2}.
\]

\textbf{Computação de $1-\sin x$ e $1-\cos x$:}
\[
1-\sin x = 1 - x + \frac{x^3}{6} + \cdots,
\]
\[
1-\cos x = \frac{x^2}{2} - \frac{x^4}{24} + \cdots \approx \frac{x^2}{2}.
\]

\textbf{Tabela completa:}
\[
\sin x \approx x - \frac{x^{3}}{6},\quad
\cos x \approx 1 - \frac{x^{2}}{2},
\]
\[
1-\sin x \approx 1 - x + \frac{x^{3}}{6},\quad
1-\cos x \approx \frac{x^{2}}{2}.
\]
