\textbf{Fórmula do período do pêndulo simples:}
\[
T=2\pi\sqrt{\frac{L}{g}}\implies T\propto\sqrt{L}.
\]

\textbf{Variação percentual:} se $L\to L'$, então $T'\propto\sqrt{L'}$, e a variação relativa é:
\[
\frac{T'-T}{T}=\frac{\sqrt{L'}-\sqrt{L}}{\sqrt{L}}=\sqrt{\frac{L'}{L}}-1.
\]

\textbf{(a) $L' = L/2$ (metade):}
\[
\frac{T'}{T}=\sqrt{\frac{1}{2}}=\frac{1}{\sqrt{2}}\approx0{,}7071.
\]
Variação: $T'\approx0{,}7071\,T$, ou seja, \textbf{diminui em $\approx29{,}3\%$}.

\textbf{(b) $L' = \frac{2}{3}L$ (reduzido em um terço):}
\[
\frac{T'}{T}=\sqrt{\frac{2}{3}}\approx0{,}8165.
\]
Variação: \textbf{diminui em $\approx18{,}4\%$}.

\textbf{(c) $L' = L/3$ (um terço):}
\[
\frac{T'}{T}=\sqrt{\frac{1}{3}}=\frac{1}{\sqrt{3}}\approx0{,}5774.
\]
Variação: \textbf{diminui em $\approx42{,}3\%$}.
