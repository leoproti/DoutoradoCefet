\textbf{Regra:} em multiplicações, o resultado tem tantos algarismos significativos quanto o fator com \emph{menos} algarismos significativos.

\begin{enumerate}
\item $6{,}28\times10^{3}$ tem 3 al. sig.; $2{,}712$ tem 4 al. sig. $\Rightarrow$ resultado com 3 al. sig.:
\[
6280 \times 2{,}712 = 17031{,}36 \approx \mathbf{1{,}70\times10^{4}}.
\]

\item $43{,}32$ tem 4 al. sig.; $0{,}3$ tem \textbf{1} al. sig. $\Rightarrow$ resultado com 1 al. sig.:
\[
43{,}32 \times 0{,}3 = 12{,}996 \approx \mathbf{10}.
\]

\item $928$ tem 3 al. sig.; $4{,}23$ tem 3 al. sig. $\Rightarrow$ resultado com 3 al. sig.:
\[
928 \times 4{,}23 = 3925{,}44 \approx \mathbf{3{,}93\times10^{3}}.
\]
\end{enumerate}
