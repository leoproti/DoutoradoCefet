\textbf{Período do pêndulo em função da gravidade:}
\[
T=2\pi\sqrt{\frac{L}{g}}\implies \frac{T_{\text{planeta}}}{T_{\text{Terra}}}=\sqrt{\frac{g_{\text{Terra}}}{g_{\text{planeta}}}}.
\]

\textbf{Dados:}
\[
g_{\text{Terra}}=9{,}80\,\mathrm{m/s^2},\quad g_{\text{Marte}}=3{,}71\,\mathrm{m/s^2},\quad g_{\text{Plutão}}=0{,}62\,\mathrm{m/s^2}.
\]

\textbf{Em Marte:}
\[
\frac{T_{\text{Marte}}}{T_{\text{Terra}}}=\sqrt{\frac{9{,}80}{3{,}71}}=\sqrt{2{,}6417}\approx1{,}625.
\]
O período em Marte é \textbf{$\approx62{,}5\%$ maior} que na Terra.

\textbf{Exemplo numérico:} se $T_{\text{Terra}}=1{,}0\,\mathrm{s}$ (pêndulo de $L=0{,}248\,\mathrm{m}$), então $T_{\text{Marte}}\approx1{,}625\,\mathrm{s}$.

\textbf{Em Plutão:}
\[
\frac{T_{\text{Plutão}}}{T_{\text{Terra}}}=\sqrt{\frac{9{,}80}{0{,}62}}=\sqrt{15{,}806}\approx3{,}976\approx4{,}0.
\]
O período em Plutão é \textbf{$\approx300\%$ maior} (quase 4 vezes) que na Terra.
